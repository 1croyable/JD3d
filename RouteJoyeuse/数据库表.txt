-- 用户表
CREATE TABLE `user` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `username` VARCHAR(255) NOT NULL UNIQUE,
    `password` VARCHAR(255) NOT NULL,
    `phone_number` VARCHAR(20),
    `privilege_level` INT DEFAULT 0
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- 文件表
CREATE TABLE `Files` (
    `file_id` INT AUTO_INCREMENT PRIMARY KEY,
    `user_id` INT NOT NULL,
    `name` VARCHAR(255) NOT NULL,
    `type` VARCHAR(50),
    `size` BIGINT,
    `create_time` DATETIME,
    `cos_key` VARCHAR(255) NOT NULL,
    FOREIGN KEY (`user_id`) REFERENCES `user`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- 文件夹表
CREATE TABLE `Folders` (
    `folder_id` INT AUTO_INCREMENT PRIMARY KEY,
    `user_id` INT NOT NULL,
    `parent_id` INT DEFAULT NULL,
    `name` VARCHAR(255) NOT NULL,
    `create_time` DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (`user_id`) REFERENCES `user`(`id`) ON DELETE CASCADE,
    FOREIGN KEY (`parent_id`) REFERENCES `Folders`(`folder_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- 用户文件夹关系表
CREATE TABLE `User_Folders` (
    `user_folder_id` INT AUTO_INCREMENT PRIMARY KEY,
    `user_id` INT NOT NULL,
    `folder_id` INT NOT NULL,
    FOREIGN KEY (`user_id`) REFERENCES `user`(`id`) ON DELETE CASCADE,
    FOREIGN KEY (`folder_id`) REFERENCES `Folders`(`folder_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- 文件夹层级结构表
CREATE TABLE `Folder_Structure` (
    `structure_id` INT AUTO_INCREMENT PRIMARY KEY,
    `parent_id` INT,
    `child_id` INT NOT NULL,
    FOREIGN KEY (`parent_id`) REFERENCES `Folders`(`folder_id`) ON DELETE SET NULL,
    FOREIGN KEY (`child_id`) REFERENCES `Folders`(`folder_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- 文件与文件夹关系表
CREATE TABLE `File_Folder_Relations` (
    `relation_id` INT AUTO_INCREMENT PRIMARY KEY,
    `file_id` INT NOT NULL,
    `folder_id` INT NOT NULL,
    FOREIGN KEY (`file_id`) REFERENCES `Files`(`file_id`) ON DELETE CASCADE,
    FOREIGN KEY (`folder_id`) REFERENCES `Folders`(`folder_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- 用户头像表
CREATE TABLE user_avatars (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    cosKey VARCHAR(255) NOT NULL,
    lastKey VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_user FOREIGN KEY (user_id) REFERENCES user(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 笔记表
CREATE TABLE `Notes` (
    `note_id` INT AUTO_INCREMENT PRIMARY KEY,
    `file_id` INT NOT NULL,
    `content` JSON,
    `create_time` DATETIME NOT NULL,
    `last_modified_time` DATETIME NOT NULL,
    FOREIGN KEY (`file_id`) REFERENCES `Files`(`file_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- 工作页表
CREATE TABLE `Work` (
    `work_id` INT AUTO_INCREMENT PRIMARY KEY,
    `user_id` INT NOT NULL,
    `data` JSON NOT NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (`user_id`) REFERENCES `user`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
